## 小美的因子数量
- 题目链接：https://www.nowcoder.com/exam/test/95484615/detail?pid=66371995&examPageSource=Company&subTabName=written_page&testCallback=https%3A%2F%2Fwww.nowcoder.com%2Fexam%2Fcompany%3FcurrentTab%3Drecommand%26jobId%3D100%26tagIds%3D179&testclass=%E8%BD%AF%E4%BB%B6%E5%BC%80%E5%8F%91

  小美很喜欢因子数量为奇数的数。
  现在小芳给了小美一个区间 [l,r][*l*,*r*]，请你帮小美算出区间内有多少个因子数量为奇数的数。

  注意，这题使用正常计算因子数量的方法会超时，有一个数学技巧：

  **因子数量为奇数的数恰好就是完全平方数，因为因子总是成对出现，只有d=n/d时才多一个。**

  **更骚的是，只需要计算左右边界的开根之间有多少个数字（注意处理边界），就能够直接得出左右边界中有多少个完全平方数**

  ~~~go
  package main
  
  import (
      "fmt"
      "math"
  )
  
  func main() {
  	left, right := 0.0, 0.0
  	fmt.Scan(&left, &right)
  	count := math.Floor(math.Sqrt(right)) - math.Floor(math.Sqrt(left-1))
  	res := int(count)
  	fmt.Printf("%d", res)
  }
  ~~~

  ~~~go
  package main
  
  import (
  	"bufio"
  	"fmt"
  	"os"
  	"strconv"
  )
  
  // 修正 MOD 的值：1e9 + 7 (即 1000000007)
  const MOD = 1000000007
  
  func main() {
  	// 1. 设置快速输入 (Fast Reader)
  	scanner := bufio.NewScanner(os.Stdin)
  	scanner.Split(bufio.ScanWords) // 按空格或换行分割
  	// 扩大 buffer 容量，防止遇到极长行时溢出（部分平台要求）
  	buf := make([]byte, 1024*1024)
  	scanner.Buffer(buf, bufio.MaxScanTokenSize)
  
  	// 2. 设置快速输出 (Fast Writer)
  	writer := bufio.NewWriter(os.Stdout)
  	// 【重点】程序结束前必须 Flush，否则输出会丢失！
  	defer writer.Flush()
  
  	// 辅助读取数字的函数
  	scanInt := func() int {
  		scanner.Scan()
  		res, _ := strconv.Atoi(scanner.Text())
  		return res
  	}
  
  	// 开始读取数据
  	if !scanner.Scan() {
  		return
  	}
  	k, _ := strconv.Atoi(scanner.Text())
  	q := scanInt()
  
  	queries := make([]int, q)
  	maxIndex := 0
  	for i := 0; i < q; i++ {
  		queries[i] = scanInt()
  		if queries[i] > maxIndex {
  			maxIndex = queries[i]
  		}
  	}
  
  	// f 表示第 i 项，pre 表示前 i 项和
  	// 容量设为 maxIndex + 1，与 C++ 代码保持一致，避免索引越界
  	f := make([]int, maxIndex+1)
  	pre := make([]int, maxIndex+1)
  
  	// 使用你 C++ 版本中安全且简洁的 DP 逻辑
  	for i := 1; i <= maxIndex; i++ {
  		if i <= k {
  			f[i] = 1
  		} else {
  			left := 0
  			if i-k-1 >= 0 {
  				left = pre[i-k-1]
  			}
  			// S_i = pre[i-1] - pre[i-k-1]
  			f[i] = (pre[i-1] - left + MOD) % MOD
  		}
  		pre[i] = (pre[i-1] + f[i]) % MOD
  	}
  
  	// 3. 将输出写入 buffer，而不是直接输出到控制台
  	for _, x := range queries {
  		fmt.Fprintln(writer, f[x])
  	}
  }
  ~~~
  
  主要记忆这个IO的输入输出方式

### 31.给定一个整数数组,判断其中是否有3个数和为N

- 题目链接：https://www.nowcoder.com/exam/test/95921350/detail?pid=16518748&examPageSource=Company&subTabName=written_page&testCallback=https%3A%2F%2Fwww.nowcoder.com%2Fexam%2Fcompany%3FquestionJobId%3D10%26subTabName%3Dwritten_page&testclass=%E8%BD%AF%E4%BB%B6%E5%BC%80%E5%8F%91

  其实就是三数之和，但是简化版，不用去重，这次使用了双指针方法，更重要的是，记住这种快速的输入输出方法

  ~~~go
  package main
  
  import (
      "fmt"
      "bufio"
      "strings"
      "strconv"
      "os"
      "sort"
  )
  
  func threeSum(nums []int, k int) bool {
  	sort.Ints(nums)
  	numLen := len(nums)
  	for ptr := 0; ptr < numLen-2; ptr++ {
  		left, right := ptr+1, numLen-1
  		for left < right {
  			sum := nums[ptr] + nums[left] + nums[right]
  			if sum < k {
  				left++
  			} else if sum > k {
  				right--
  			} else {
  				return true
  			}
  		}
  	}
  	return false
  }
  
  func main() {
  	scanner := bufio.NewScanner(os.Stdin)
  	buf := make([]byte, 1024*1024)
  	scanner.Buffer(buf, bufio.MaxScanTokenSize)
  
  	writer := bufio.NewWriter(os.Stdout)
  	defer writer.Flush()
  
  	splitFunc := func(c rune) bool {
  		return c == ' ' || c == ',' || c == '\t'
  	}
  	nums := []int{}
  	for scanner.Scan() {
  		line := scanner.Text()
  		if strings.TrimSpace(line) == "" {
  			continue
  		}
  		tokens := strings.FieldsFunc(line, splitFunc)
  		for _, c := range tokens {
  			val, _ := strconv.Atoi(c)
  			nums = append(nums, val)
  		}
  	}
  	k := nums[len(nums)-1]
  	nums = nums[:len(nums)-1]
  	var res string
  	if threeSum(nums, k) {
  		res = "True"
  	} else {
  		res = "False"
  	}
  	fmt.Fprintln(writer, res)
  }
  ~~~
