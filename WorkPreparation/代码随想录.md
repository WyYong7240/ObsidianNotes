  ~~~go
  	"bufio"

  func jsonInput() {

  		return

  }
  ~~~

  ~~~go
  	var matrix [][]int
  
  	for scanner.Scan() {
  		strs := strings.Split(scanner.Text(), " ")
  		var row []int
  		for _, s := range strs {
  			num, err := strconv.Atoi(s)
  			if err != nil {
  				fmt.Printf("Invalid Input: %s\n", s)
  				return
  			}
  			row = append(row, num)
  		}
  		matrix = append(matrix, row)
  	}
  	fmt.Printf("Receive 3D Array: %v\n", matrix)
  }
  ~~~

   ~~~go
   func ShellSort(num *[]int) {
   	delta := len(*num) / 2
   	for delta != 0 {
   		for i := delta; i < len(*num); i++ {
   			if (*num)[i] < (*num)[i-delta] {
   				tmp := (*num)[i]
   				j := i - delta
   				for ; j >= 0 && (*num)[j] > tmp; j -= delta {
   					(*num)[j+delta] = (*num)[j]
   				}
   				(*num)[j+delta] = tmp
   			}
   		}
   		delta /= 2
   	}
   }
   
   func main() {
   	num := []int{36, 18, 10, 2, 88, 53, 27, 99, 0, 38, 46, 77, 78, 55, 40}
   	ShellSort(&num)
   	fmt.Println(num)
   }
   ~~~

4. 性能分析

   空间效率：$O(1)$

   时间效率：希尔排序的时间复杂度不方便计算，n在某个特定范围是，时间复杂度为$O(n^{1.3})$，最坏为$O(n^2)$

   稳定性：当相同关键字的记录被划分到不同的子表时，可能会改变其相对次序，因而希尔排序不稳定

   ~~~go
   func HeapSort(num *[]int) {
   	length := len(*num)
   	// 从最后一个非叶节点开始调整，直至成为一个小根堆
   	for i := length/2 - 1; i >= 0; i-- {
   		HeapAdjust(num, i, length)
   	}
   	for i := length-1; i >= 1; i-- {
   		// 将小根弹出，放到最后一个元素，将最后一个元素放到堆顶，重新调整
   		tmp := (*num)[0]
   		(*num)[0] = (*num)[i]
   		(*num)[i] = tmp
   
   		HeapAdjust(num, 0, i)
   	}
   }
   
   func HeapAdjust(num *[]int, s, m int) {
   	rc := (*num)[s]
   	for j := s*2+1; j < m; j = j*2+1 {
   		// j+1,j+2分别是s的左右孩子，选取较小的那个与根s比较
   		if j+1 < m && (*num)[j] > (*num)[j+1] {
   			j++
   		}
   		// 如果根元素小于左右子树中最小的那个，这个子树具有小根堆性质，调整完毕
   		if rc < (*num)[j] {
   			break
   		} else {
   			// 否则将较小元素与根进行交换，并将根标记换位成较小子树的根节点编号，递归调整
   			(*num)[s] = (*num)[j]
   			s = j
   		}
   	}
   	// 将根原本根s的值放到最终调整位置
   	(*num)[s] = rc
   }
   
   func main() {
   	num := []int{36, 18, 10, 2, 88, 53, 27, 99, 0, 38, 46, 77, 78, 55, 40}
   	// ShellSort(&num)
   	HeapSort(&num)
   	fmt.Println(num)
   }
   ~~~

   * 空间效率：空间复杂度为$O(1)$

   * 时间效率：建堆时间是$O(n)$，之后有n-1次向下调整操作，每次调整的时间复杂度是$O(h)$，故最好，最坏，平均时间复杂度为$O(nlog_2n)$

     > 完全二叉树n个结点，$\lfloor n/2 \rfloor$ 个结点是非叶结点，每个非叶结点下降调整时有两次比较，而第i层的结点最多下降h-i层，最多2(h-i)次比较，第i层最多有$2^{i-1}$个结点，故调整总的比较次数为$\sum\limits_{i=1}^{n-1}2^{i-1}*2(h-i)$故，建堆的复杂度为$O(n)$

   ~~~go 
   func quickSortPartition(num *[]int, left, right int) int {
   	pivotKey := (*num)[left]
   	i, j := left, right
   	for i < j {
   		for i < j && (*num)[j] >= pivotKey {
   			j--
   		}
   		(*num)[i] = (*num)[j]
   		for i < j && (*num)[i] <= pivotKey {
   			i++
   		}
   		(*num)[j] = (*num)[i]
   	}
   	(*num)[i] = pivotKey
   	return i
   }
   
   func quickSort(num *[]int, left, right int) {
   	if left < right {
   		pivotLoc := quickSortPartition(num, left, right)
   		quickSort(num, left, pivotLoc-1)
   		quickSort(num, pivotLoc+1, right)
   	}
   }
   
   func main() {
   	num := []int{36, 18, 10, 2, 88, 53, 27, 99, 0, 38, 46, 77, 78, 55, 40}
   	// ShellSort(&num)
   	// HeapSort(&num)
   	quickSort(&num, 0, len(num)-1)
   	fmt.Println(num)
   }
   ~~~

## 归并排序

* 排序过程

  > 2路归并的归并树是倒悬着的二叉树，二叉树第h层最多$2^{h-1}$个结点，有$n \le 2^{h-1}得出h-1 = \lceil log_2n \rceil$ h-1为归并趟数

  ![image-20230928221445955](https://typora2icture.oss-cn-beijing.aliyuncs.com/Algorithm/image-20230928221445955.png)

* 代码

  ~~~go
  func Merge(num []int, left, right, mid int) {
  	// 将待合并的数组元素移出来，然后才能动原本数组中的元素
  	backArray := make([]int, right-left+1)
  	copy(backArray, num[left:right+1])
  	// 由于复制到另一个数组后，下标就改变了，所以-left，变为相对下标
  	i, j, k := 0, mid+1-left, left
  
  	for ; i <= mid-left && j <= right-left; k++ {
  		// 选取两段数组中元素较小的那个元素放到原数组中
  		if backArray[i] <= backArray[j] {
  			num[k] = backArray[i]
  			i++
  		} else {
  			num[k] = backArray[j]
  			j++
  		}
  	}
  	for i <= mid-left {
  		num[k] = backArray[i]
  		i++
  		k++
  	}
  	for j <= right-left {
  		num[k] = backArray[j]
  		j++
  		k++
  	}
  }
  
  func MergeSort(num []int, left, right int) {
  	if left < right {
  		mid := (left + right) / 2
  		MergeSort(num, left, mid)
  		MergeSort(num, mid+1, right)
  		Merge(num, left, right, mid)
  	}
  }
  
  func main() {
  	num := []int{36, 18, 10, 2, 88, 53, 27, 99, 0, 38, 46, 77, 78, 55, 40}
  	// ShellSort(&num)
  	// HeapSort(&num)
  	// quickSort(&num, 0, len(num)-1)
  	MergeSort(num, 0, len(num)-1)
  	fmt.Println(num)
  }
  ~~~
  
* 性能分析

  空间复杂度为$O(n)$

  时间效率：每趟归并的时间复杂度为$O(n)$，共$\lceil log_2n \rceil$趟归并，时间复杂度为$O(nlog_2n)$

  merge操作不会改变相同关键字记录的相对次序，因而稳定


# 数组

## 二分查找

> 二分查找的前提，数组为有序数组，并且数组中无重复元素；如果有重复元素，二分查找的返回下标就不是唯一的了。
>
> **写二分法，区间的定义一般为两种，左闭右闭即[left, right]，或者左闭右开即[left, right)。**

### **二分法的第一种写法**

* 第一种写法定义target是在一个左闭右闭的区间里，**也就是[left, right]**，[具体链接](https://www.programmercarl.com/0704.%E4%BA%8C%E5%88%86%E6%9F%A5%E6%89%BE.html#%E7%AE%97%E6%B3%95%E5%85%AC%E5%BC%80%E8%AF%BE:~:text=%E5%8C%BA%E9%97%B4%E7%9A%84%E5%AE%9A%E4%B9%89%E8%BF%99,%E4%BD%8D%E7%BD%AE%E5%B0%B1%E6%98%AF%20middle%20%2D%201)
  **因为定义在一个两边都是闭区间，所以区间两边的值也是要搜索的，所以left=right是有意义的**
  即当left=right的时候，中间的值也是要检查的
  当`if (nums[middle] > target) right`时，赋值为 middle - 1，**因为当前这个nums[middle]一定不是target，那么接下来要查找的左区间结束下标位置就是 middle - 1**

* 题目链接：https://leetcode.cn/problems/binary-search/description/

  ~~~GO
  // 二分查找的第一种写法， 左右都是闭区间定义
  func search(nums []int, target int) int {
  	var left int = 0
  	var right int = len(nums) - 1
  	for left <= right {
  		middle := left + (right-left)/2
  		if nums[middle] == target {
  			return middle
  		} else if nums[middle] > target {
  			right = middle - 1
  		} else {
  			left = middle + 1
  		}
  	}
  	return -1
  }
  ~~~

  时间复杂度$O(logn)$

  空间复杂度$O(1)$

### 二分法的第二种写法

* 如果定义target在左闭右开的区间中，也就是[left, right)

  `while(left<right)`,**因为left==right在区间[left, right)是没有意义的**

  `if (nums[middle] > target) right` 更新为 middle，**因为当前nums[middle]不等于target，去左区间继续寻找，而寻找区间是左闭右开区间，所以right更新为middle，即：下一个查询区间不会去比较nums[middle]**

  需要注意的是，除了循环条件和边界设定，初始的边界设定`right`也不一样，左闭右开的初始边界定义`right=len(nums)`这样的话，初始搜索空间才包括整个数组，如果为`len(nums)-1`就少了一个元素不在搜索空间中

* 题目链接：https://leetcode.cn/problems/binary-search/description/

  ~~~GO
  // 二分查找的第二种写法，左闭右开区间
  func search(nums []int, target int) int {
  	var left int = 0
  	var right int = len(nums)
  	for left < right {
  		middle := left + (right-left)/2
  		if nums[middle] == target {
  			return middle
  		} else if nums[middle] > target {
  			right = middle
  		} else {
  			left = middle + 1
  		}
  	}
  	return -1
  }
  ~~~

## 移除链表元素

* 题目链接：https://leetcode.cn/problems/remove-linked-list-elements/description/

  题目给出的是无dummy Head的链表头节点，只需要自己构建一个dummyHead再删除就好删除了

  ~~~go
  func removeElements(head *ListNode, val int) *ListNode {
      dummyHead := ListNode {
          Val: 0,
          Next: head,
      }
  
      pre := &dummyHead
      cur := pre.Next
      for cur != nil {
          if cur.Val == val {
  ~~~

  ~~~go
  	ptrA := headA
  	ptrB := headB
  	lenA := 0
  	lenB := 0
  	for ptrA != nil {
  		ptrA = ptrA.Next
  		lenA++
  	}
  	for ptrB != nil {
  		ptrB = ptrB.Next
  		lenB++
  	}

  	ptrB = headB
  	if lenA > lenB {
  		for i := 0; i < lenA-lenB; i++ {
  			ptrA = ptrA.Next
  		}
  	} else {
  		for i := 0; i < lenB-lenA; i++ {
  			ptrB = ptrB.Next
  		}
  	}
  	for ptrA != nil && ptrB != nil {
  		if ptrA == ptrB {
  			return ptrA
  		}
  		ptrA = ptrA.Next
  		ptrB = ptrB.Next
  	}
  	return nil
  }
  ~~~

  2. 遍历大A和大B数组，统计两个数组元素之和，和出现的次数，放到map中。
  3. 定义int变量count，用来统计 a+b+c+d = 0 出现的次数。

  5. 最后返回统计值 count 就可以了

  ~~~go
  func fourSumCount(nums1 []int, nums2 []int, nums3 []int, nums4 []int) int {

  	for _, num1 := range nums1 {
  		for _, num2 := range nums2 {
  			m[num1+num2]++
  		}
  	}
  	sum := 0
  	for _, num1 := range nums3 {
  		for _, num2 := range nums4 {
  			target := -(num1 + num2)
  			if v, ok := m[target]; ok {
  				sum += v
  			}
  		}
  	}
  	return sum
  ~~~

## 三数之和

* 题目链接：https://leetcode.cn/problems/3sum/

  ~~~go
  
  func threeSum(nums []int) [][]int {
      ans := [][]int{}
  
      // 用于将答案进行去重，去重key是排序完成的3元素数组
      curRes := make(map[[3]int]bool)
  
      for i, num := range nums {
          //存放当前轮已经遍历过的元素
          m := make(map[int]int)
          for j := i + 1; j < len(nums); j++ {
              num1, num2 := num, nums[j]
              target := -(num1 + num2)
              // 如果目标元素在该轮次以及遍历过，那就构造一个可能答案
              if _, ok := m[target]; ok {
                  tmpAns := []int{num1, num2, target}
                  // 对可能答案进行排序去重，如果排序完成后，可能答案已经在之前出现过，丢弃，否则采用
                  sort.Ints(tmpAns)

                      ans = append(ans, tmpAns)
                      curRes[[3]int(tmpAns)] = true
                  }
              } else {
                  // 如果目标元素没有遍历过，那就将本次遍历的元素加入map

              }
          }
      }
      return ans
  }
  ~~~

  ~~~go
  class Solution {
  public:
      // 在一个数组中找到3个数形成的三元组，它们的和为0，不能重复使用（三数下标互不相同），且三元组不能重复。
      // b（存储）== 0-(a+c)（检索）
      vector<vector<int>> threeSum(vector<int>& nums) {
          vector<vector<int>> result;
          sort(nums.begin(), nums.end());
          
          for (int i = 0; i < nums.size(); i++) {
              // 如果a是正数，a<b<c，不可能形成和为0的三元组
              if (nums[i] > 0)
                  break;
              
              // [a, a, ...] 如果本轮a和上轮a相同，那么找到的b，c也是相同的，所以去重a
              if (i > 0 && nums[i] == nums[i - 1])
                  continue;
              
              // 这个set的作用是存储b
              unordered_set<int> set;
              
              for (int k = i + 1; k < nums.size(); k++) {
                  // 去重b=c时的b和c
                  if (k > i + 2 && nums[k] == nums[k - 1] && nums[k - 1] == nums[k - 2])
                      continue;
                  
                  // a+b+c=0 <=> b=0-(a+c)
                  int target = 0 - (nums[i] + nums[k]);
                  if (set.find(target) != set.end()) {
                      result.push_back({nums[i], target, nums[k]});   // nums[k]成为c
                      set.erase(target);
                  }
                  else {
                      set.insert(nums[k]);                            // nums[k]成为b
                  }
              }
          }
  
          return result;
      }
  };
  ~~~

  双指针法：
  
  ~~~go
  
  func threeSum(nums []int) [][]int {
      ans := [][]int{}
      n := len(nums)
      sort.Ints(nums)
  
      // 开始枚举第一个数，由于对数组排过序，为了去重，要求该轮次第一个数和上一轮次第一个数不一致
      for first := 0; first < n; first++ {
          if first > 0 && nums[first] == nums[first-1] {
              continue
          }
  
          // 默认第三个数从剩余数组的最右端开始
          third := n-1
          // 第二个数和第三个数之和要求为 -nums[first]
          target := -1 * nums[first]
          // 开始枚举第二个数，为了去重，同样要求和上一轮第二个数不一致
          for second := first + 1; second < n; second++ {
              if second > first + 1 && nums[second] == nums[second-1] {
                  continue
              }
              // 同时保证第二个数的指针在第三个数的左侧;如果第二个数+第三个数大于目标值，由于排序过，所以需要第三个数的指针向左移动，变小
              for second < third && nums[second] + nums[third] > target {
                  third--
              }
              // 如果变小的同时，指针重合，表示本轮没有可用答案
              if second == third {
                  break
              }
              // 如果指针没有重合，并且第二第三数之和没有比target大了，校验一下是否等于target，等于就可以作为答案之一了
              if nums[second] + nums[third] == target {
                  ans = append(ans, []int{nums[first], nums[second], nums[third]})
              }
          }
      }
      return ans
  }
  ~~~

  ~~~go
  	}
  ~~~

  ~~~go
  		if i+k-1 > len(str)-1 {
  			reverseStringIndex(str, i, len(str)-1)
  			break
  		} else if i+2*k > len(str)-1 {
  			reverseStringIndex(str, i, i+k-1)
  			break
  		} else {
  			reverseStringIndex(str, i, i+k-1)
  		}
  	}
  	return string(str)
  }
  
  func reverseStringIndex(str []rune, from, to int) {
  	left := from
  	right := to
  	for left < right {
  		tmp := str[left]
  		str[left] = str[right]
  		str[right] = tmp
  		left++
  		right--
  	}
  }
  ~~~

  ~~~go
  func reverseStr(s string, k int) string {
      ss := []byte(s)
      length := len(s)
      for i := 0; i < length; i += 2 * k {
       // 1. 每隔 2k 个字符的前 k 个字符进行反转
       // 2. 剩余字符小于 2k 但大于或等于 k 个，则反转前 k 个字符
          if i + k <= length {
              reverse(ss[i:i+k])
          } else {
              reverse(ss[i:length])
          }
      }
      return string(ss)

      left := 0
      right := len(b) - 1
      for left < right {
          b[left], b[right] = b[right], b[left]
          left++
          right--
  ~~~

  ~~~go
  func merge(intervals [][]int) [][]int {
      ans := [][]int{}
      // 对二维数组进行自定义排序
      sort.Slice(intervals, func(i, j int) bool {
          if intervals[i][0] < intervals[j][0] {
              return true
          } else if intervals[i][0] == intervals[j][0] {
              return intervals[i][1] <=intervals[j][1]
          }
          return false
      })
  
      // 当前处理的数组
      cur := intervals[0]
      n := len(intervals)
      for i := 1; i < n; i++ {
          // 如果待处理的数组的开头小预等于当前数组的结尾
          if intervals[i][0] <= cur[1] {
              // 有两种情况，待处理的数组的结尾可能也小于当前处理数组的结尾，所以要取max
              cur[1] = max(intervals[i][1], cur[1])
          } else {
              // 否则，当前处理数组已经合并完毕，开始处理下一个
              ans = append(ans, cur)
              cur = intervals[i]
          }
      }
      ans = append(ans, cur)
      return ans
  }
  ~~~


  总体思路和前缀和很像，和之前的接雨水的思想也很像，统计左右两侧的乘积，然后分别使用两个数组的内容进行当前位置的计算
  
  ~~~go
  func productExceptSelf(nums []int) []int {
      length := len(nums)
  
      // L 和 R 分别表示左右两侧的乘积列表
      L, R, answer := make([]int, length), make([]int, length), make([]int, length)
  
      // L[i] 为索引 i 左侧所有元素的乘积
      // 对于索引为 '0' 的元素，因为左侧没有元素，所以 L[0] = 1
      L[0] = 1
      for i := 1; i < length; i++ {
          L[i] = nums[i-1] * L[i-1]
      }
  
      // R[i] 为索引 i 右侧所有元素的乘积
      // 对于索引为 'length-1' 的元素，因为右侧没有元素，所以 R[length-1] = 1
      R[length-1] = 1
      for i := length - 2; i >= 0; i-- {
          R[i] = nums[i+1] * R[i+1]
      }
      
      // 对于索引 i，除 nums[i] 之外其余各元素的乘积就是左侧所有元素的乘积乘以右侧所有元素的乘积
      for i := 0; i < length; i++ {
          answer[i] = L[i] * R[i]
      }
      return answer
  }
  ~~~

### 128. 最长连续序列

* 题目链接：https://leetcode.cn/problems/longest-consecutive-sequence/description/?envType=study-plan-v2&envId=top-100-liked

  ~~~go
  func longestConsecutive(nums []int) int {
      numSet := map[int]bool{}
      for _, num := range nums {
          numSet[num] = true
      }
      ans := 0
      for key, _ := range numSet {
          if !numSet[key-1] {
              // 如果这个比这个数小1的数不存在，即这个数已经是某个子序列中最小的数了，从这个数开始算
              curNum := key
              lenCount := 1
              // 从这个子序列最小的数开始向大了找，计算该子序列的长度
              for numSet[curNum+1] {
                  curNum ++
                  lenCount++
              }
              if lenCount > ans {
                  ans = lenCount
              }
          }
      }
      return ans
  }
  ~~~

  其实这题最重要的思想就在于，如何避免重复计算，4，3，2，1都在数组中，如果3也要计算连续长度，那不合适，需要找到连续数组的头，要么只有4计算，要么只有1计算
  
  **最精华在于如果这个数的+1或者-1存在数组中，那么这个数不计算连续长度**

  ~~~go
  		}
  ~~~

### 3. 无重复字符的最长子串

- 题目链接：https://leetcode.cn/problems/longest-substring-without-repeating-characters/description/?envType=study-plan-v2&envId=top-100-liked

  依旧滑动窗口，主要思想就是快慢指针+去重用的结构；慢指针初始为0，快指针从0开始，如果fast字母在结构中已经有了，看当前子串长度，然后看上一个该元素的位置，新字串从上一个该元素的位置开始算，所以要把本次子串开始的位置到上一个该字母的位置的元素从去重结构中删除，开始新子串的长度计算

  ~~~go
  
  
  func lengthOfLongestSubstring(s string) int {
      m := map[rune]int{}
      str := []rune(s)
      // start表示当前子串的开始下标
      start, ans := 0, 0
  
      for i, c := range str {
          // 如果当前字符，在当前字串之前没出现过，加入map
          if index, ok := m[c]; !ok {
              m[c]=i
          } else {
              // 如果当前字符出现在过当前字串，计算当前字串的长度，并取最大
              ans = max(i-start, ans)
              // 然后将当前子串的开始位置，移动到该重复字符上一次所在位置
              // 并将更新后不在当前子串中的字符从map中删除
              for start <= index {
                  delete(m, str[start])
                  start++
              }
              // 最后，更新该字符的最新位置
              m[c]=i
          }
      }
      // 还有最后一段子串因为无法触发重复字符，导致无法计算，需要单独计算
      ans = max(len(s)-start, ans)
      return ans
  }
  ~~~

  ~~~go
          for q.IntSlice[0] <= i-k {
              heap.Pop(q)
          }
          // 当前滑动窗口中的最大元素
          ans = append(ans, nums[q.IntSlice[0]])
      }
      return ans
  }
  
  // 实际存储数组数据
  var a []int
  type hp struct {
      // 建堆，使用默认会排序的结构
      // 存储的是数组元素下标
      sort.IntSlice
  }
  // 堆结构默认比较大小的方式
  func (h *hp)Less(i, j int) bool {
      return a[h.IntSlice[i]] > a[h.IntSlice[j]]
  }
  func (h *hp)Push(v interface{}) {
      h.IntSlice = append(h.IntSlice, v.(int))
  }
  func (h *hp)Pop() interface{} {
      a := h.IntSlice
      v := a[len(a)-1]
      h.IntSlice = a[:len(a)-1]
      return v
  }
  ~~~

  解答如下几个问题

  1. 为什么原算法使用sort.IntSlice，本质上是一个[]int数组，但是这个类型自己实现了heap要求的Less、Len、Swap方法，如果替换为普通数组，还需要自己实现额外的两个函数
  2. 原本Pop中的a和全局变量a不是一个a，这在新的代码中有体现
  3. 原算法先给堆添加元素再init堆，这是将这个数组开启堆模式；也可以先init堆，然后使用heap(q, i)来初始化前K个元素，但是这样效率会慢

### 76. 最小覆盖子串

* 题目链接：https://leetcode.cn/problems/minimum-window-substring/description/?envType=study-plan-v2&envId=top-100-liked

  其实本质思路就是双指针加两个特征HasMap做匹配，只要当前子串满足要求，就右移子串的起点，否则右移子串的终点

  ~~~go
  func minWindow(s string, t string) string {
      // 初始化模板字串的map特征
      tMap := map[byte]int{}
      for _, c := range t {
          tMap[byte(c)]++
      }
  
      // 定义并初始化当前子串的特征map
      curMap := map[byte]int{}
      // 当前子串的左右下标
      start, end, n := 0, 0, len(s)
      curMap[s[end]]++
      // ans其实可以初始化为一个长度大于s的任意字符串
      ans := s + "/"
      for end < n {
          // 当前子串的特征是否匹配
          if ok := isEqual(tMap, curMap); ok {
              // 如果匹配，右移子串的起始位置，并改变当前子串的特征Map
              // 先进行结果判定，如果当前特征匹配的子串长度小于当前最小子串
              if (end-start+1) < len(ans) {
                  ans = s[start:end+1]
              }
              curMap[s[start]]--
              start++
          } else {
              // 如果特征不匹配，右移子串的终点位置，并修改当前子串的特征Map
              end++
              if end < n {
                  curMap[s[end]]++
              }
          }
      }
      if ans == s + "/" {
          return ""
      }
      return ans
  }
  
  // 判定当前子串是否满足特征匹配
  func isEqual(tMap, curMap map[byte]int) bool {
      for k, v := range tMap {
          if curV, ok := curMap[k]; !ok || curV < v {
              return false
          }
      }
      return true
  }
  ~~~

### 560.和为K的子数组

  ~~~go
  func subarraySum(nums []int, k int) int {
      res, presum := 0, 0
      m := map[int]int{}
      m[0]=1
      for i := 0; i < len(nums); i++ {
          // 计算当前位置的前缀和
          presum += nums[i]
          // 只要之前子数组中，出现过和为presume-k的前缀子数组，那么那个子数组的结尾到当前位置就是子数组和为K
          if value, ok := m[presum-k]; ok {
              res += value
          }
          // 将当前前缀子数组和加入map
          m[presum] += 1
      }
      return res
  }
  ~~~

  ~~~go
   *     Val int

   * }

  func addTwoNumbers(l1 *ListNode, l2 *ListNode) *ListNode {
      preSum := 0
      ptr1 := l1
      ptr2 := l2
      dummyHead := ListNode{}
      cur := &dummyHead
      for ptr1 != nil || ptr2 != nil || preSum != 0 {
          curSum := 0
          if ptr1 != nil {
              curSum += ptr1.Val
              ptr1 = ptr1.Next
          }
          if ptr2 != nil {
              curSum += ptr2.Val
              ptr2 = ptr2.Next
          }

          node := ListNode{
              Val: curSum % 10,
              Next: nil,
          }
          cur.Next = &node
          cur = &node
          if curSum >= 10 {
              preSum = 1
          } else {
              preSum = 0
          }
      }
      return dummyHead.Next
  }
  ~~~

  ~~~go 
   *     Val int
   *     Left *TreeNode

   * }
   */
  ~~~

  ~~~go
      _, len := maxDepth(root)
      return len
  }
  
  func maxDepth(root *TreeNode) (int,int) {
      if root == nil {
          return 0,0
      }
      leftDepth, leftLen := maxDepth(root.Left)
      rightDepth, rightLen := maxDepth(root.Right)
      curLen := leftDepth + rightDepth
      maxLen := max(leftLen, rightLen, curLen)
      maxDepth := max(leftDepth, rightDepth)+1
      return maxDepth, maxLen
  }
  ~~~

  ~~~go
   * type TreeNode struct {
   *     Val int
   *     Left *TreeNode
   *     Right *TreeNode
   * }
   */
  
  func buildTree(preorder []int, inorder []int) *TreeNode {
      if len(preorder) == 0 {
          return nil
      }
  
      root := TreeNode {
          Val: preorder[0],
          Left: nil,
          Right: nil,
      }
      inorderIndex := numsFind(inorder, preorder[0])
      leftInorder := inorder[:inorderIndex]
      rightInorder := inorder[inorderIndex+1:]
  
      preorderIndex := len(preorder) - len(rightInorder)
  
      leftPreorder := preorder[1:preorderIndex]
      rightPreorder := preorder[preorderIndex:]
      
      root.Left = buildTree(leftPreorder, leftInorder)
      root.Right = buildTree(rightPreorder, rightInorder)
      return &root
  }
  
  func numsFind(nums []int, target int) int {
      for i, num := range nums {
          if num == target {
              return i
          }
      }
      return -1
  } 
  ~~~

### 437. 路径总和III

- 题目链接：https://leetcode.cn/problems/path-sum-iii/description/?envType=study-plan-v2&envId=top-100-liked

  其实有两种思路，第一种是从上到下，map中放root.Val-targetSum，如果左右孩子中有等于该值的，计数值+1；否则对map中每个元素-root.Val

  第二种，从下向上计算，如果root.Val==targetSum，计数值+1；如果左右子树不同路径求和值+root.Val\==targetSum，计数值+1

  ~~~go
  /**
   * Definition for a binary tree node.
   * type TreeNode struct {
   *     Val int
   *     Left *TreeNode
   *     Right *TreeNode
   * }
   */
  func pathSum(root *TreeNode, targetSum int) int {
  	if root == nil {
  		return 0
  	}
  	_, counter := childPath(root, targetSum)
  	return counter
  }
  
  func childPath(root *TreeNode, targetSum int) ([]int, int) {
  	if root == nil {
  		return nil, 0
  	}
  	res := []int{root.Val}
  	leftSlice, leftCounter := childPath(root.Left, targetSum)
  	rightSlice, rightCounter := childPath(root.Right, targetSum)
  
  	counter := leftCounter + rightCounter
      if root.Val == targetSum {
          counter++
      }
  	for _, num := range leftSlice {
  		if num+root.Val == targetSum {
  			counter++
  		}
  		res = append(res, num+root.Val)
  	}
  	for _, num := range rightSlice {
  		if num+root.Val == targetSum {
  			counter++
  		}
  		res = append(res, num+root.Val)
  	}
  	return res, counter
  }
  ~~~

### 236. 二叉树的最近公共祖先

- 题目链接：https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-tree/description/?envType=study-plan-v2&envId=top-100-liked

  其实就是判断，左右子树是否存在p或者q，如果左右只有一个不为空，说明只存在一个，如果两个不为空，说明该节点就是公共祖先；就是注意处理当前节点为p或者q的情况，相当于除了左右子树，还有一个中间的情况要处理

  ~~~go
  /**
   * Definition for a binary tree node.
   * type TreeNode struct {
   *     Val int
   *     Left *TreeNode
   *     Right *TreeNode
   * }
   */
  func lowestCommonAncestor(root, p, q *TreeNode) *TreeNode {
      res, _ := childFunc(root, p, q)
      return res
  }
  
  func childFunc(root, p, q *TreeNode) (*TreeNode, bool) {
      if root == nil {
          return nil, false
      }
      var cur *TreeNode
      if root == p || root == q {
          cur = root
      }
  
      left, lF := childFunc(root.Left, p, q)
      if lF {
          return left, lF
      }
      right, rF := childFunc(root.Right, p, q)
      if rF {
          return right, rF
      }
      if left == nil && right == nil && cur == nil {
          return nil, false
      }
      if (left != nil && right != nil) || (left != nil && cur != nil) || (right != nil && cur != nil) {
          return root, true
      }
      return root, false
  
  }
  ~~~

### 42.接雨水

* 题目连接：

  题目多种方法，这里采用动态规划的方式

  这个面积，本质上是找当前位置的左右边界的最大高度中那个高度较低的值，直接定义两个数组，分别存储当前位置的左右边界最大高度，然后计算面积时，选取其中较小的那个边界的高度，减去当前位置的高度，以底宽为1，面积累加即可

  ~~~go
  func trap(height []int) int {
      n := len(height)
      if n == 0 {
          return 0
      }
  
      // 同时定义该位置的左右两边的最大值，因为该位置的面积，由左右边界中较小的那个边界决定
      // 初始化左右边界
      leftMax := make([]int, n)
      leftMax[0] = height[0]
      rightMax := make([]int, n)
      rightMax[n-1] = height[n-1]
  
      // cur位置的左边界的最大值
      for i := 1; i < n; i++ {
          leftMax[i] = max(leftMax[i-1], height[i])
      }
      // cur位置的右边界的最大值
      for i := n-2; i >= 0; i-- {
          rightMax[i] = max(rightMax[i+1], height[i])
      }
  
      // 开始计算面积，底宽为1，高度为左右边界中较小的边界高度 - cur位置的高度
      ans := 0
      for i := 0; i < n; i++ {
          square := 1 * (min(leftMax[i], rightMax[i]) - height[i])
          ans += square
      }
      return ans
  }
  ~~~

  


  ### 53.最大子数组和

  - 题目链接：https://leetcode.cn/problems/maximum-subarray/description/?envType=study-plan-v2&envId=top-100-liked

    假设 nums 数组的长度是 n，下标从 0 到 n−1。

    我们用 f(i) 代表以第 i 个数结尾的「连续子数组的最大和」，那么很显然我们要求的答案就是：

    $max_{0≤i≤n−1}{f(i)}$

    因此我们只需要求出每个位置的 f(i)，然后返回 f 数组中的最大值即可。那么我们如何求 f(i) 呢？我们可以考虑 nums[i] 单独成为一段还是加入 f(i−1) 对应的那一段，这取决于 nums[i] 和 f(i−1)+nums[i] 的大小，我们希望获得一个比较大的，于是可以写出这样的动态规划转移方程：

    f(i)=max{f(i−1)+nums[i],nums[i]}

    不难给出一个时间复杂度 O(n)、空间复杂度 O(n) 的实现，即用一个 f 数组来保存 f(i) 的值，用一个循环求出所有 f(i)。考虑到 f(i) 只和 f(i−1) 相关，于是我们可以只用一个变量 pre 来维护对于当前 f(i) 的 f(i−1) 的值是多少，从而让空间复杂度降低到 O(1)，这有点类似「滚动数组」的思想。

    ~~~go
    func maxSubArray(nums []int) int {
        max := nums[0]
        for i := 1; i < len(nums); i++ {
            if nums[i] + nums[i-1] > nums[i] {
                nums[i] += nums[i-1]
            }
            if nums[i] > max {
                max = nums[i]
            }
        }
        return max
    }
    ~~~

    另一个写法
    
    ~~~go
    func maxSubArray(nums []int) int {
        // 表示当前连续子数组的和
        cur, n := nums[0], len(nums)
        // 表示历史所有子数组和的最大值
        ans := nums[0]
        for i := 1; i < n; i++ {
            // 当前连续子数组的和取当前数组元素值和之前连续子数组和+当前元素的最大值
            cur = max(cur+nums[i], nums[i])
            ans = max(ans, cur)
        }
        return ans
    }
    ~~~
    
    
    
    **其实就是判断当前元素，与前面元素之和相加后，是更大还是更小，如果更大，那就加入前面的数组，如果小了，那就以当前元素为新子数组开头**

### 1143. 最长公共子序列

* 题目链接：https://leetcode.cn/problems/longest-common-subsequence/description/

  具体的思路见题解，很好理解

  ~~~go
  func longestCommonSubsequence(text1 string, text2 string) int {
      n, m := len(text1), len(text2)
      // 创建二维DP数组
      dp := make([][]int, n+1)
      for i := 0; i < n+1; i++ {
          dp[i] = make([]int, m+1)
      }
  
      // 开始进行DP，详细思想见题解
      for i := 1; i < n+1; i++ {
          for j := 1; j < m+1; j++ {
              if text1[i-1] == text2[j-1] {
                  dp[i][j] = dp[i-1][j-1]+1
              } else {
                  dp[i][j] = max(dp[i-1][j], dp[i][j-1])
              }
          }
      }
      return dp[n][m]
  }
  ~~~



### 279.完全平方数

* 题目链接：https://leetcode.cn/problems/perfect-squares/description/?envType=study-plan-v2&envId=top-100-liked

  具体思路可以见代码也可以见题解
  
  ~~~go
  func numSquares(n int) int {
      dp := make([]int, n+1)
      // 枚举从1到n，每个数最少需要多少个平方数组成
      for i := 1; i <= n; i++ {
          // 表示当前数的i-j*j部分，最少需要多少个平方数组成
          minCount := math.MaxInt32
          for j := 1; j*j <= i; j++ {
              minCount = min(minCount, dp[i-j*j])
          }
          // 这个数的组成，可以由j*j与，i-j*j的最小组成构成，也就是1个数+另外一个数的最小组成
          dp[i] = minCount + 1
      }
      return dp[n]
  }
  ~~~
